cmake_minimum_required(VERSION 3.20)
project(PlannerAI VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(planner_core src/planner.cpp src/analysis.cpp src/calendar.cpp)
target_include_directories(planner_core PUBLIC include)
target_compile_options(planner_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(planner src/main.cpp)
target_link_libraries(planner PRIVATE planner_core)
enable_testing()
add_executable(planner_tests tests/planner_tests.cpp)
target_link_libraries(planner_tests PRIVATE planner_core)
add_test(NAME planner_tests COMMAND planner_tests)


add_executable(analysis_tests tests/analysis_tests.cpp)
target_link_libraries(analysis_tests PRIVATE planner_core)
add_test(NAME analysis_tests COMMAND analysis_tests)

add_executable(calendar_tests tests/calendar_tests.cpp)
target_link_libraries(calendar_tests PRIVATE planner_core)
add_test(NAME calendar_tests COMMAND calendar_tests)
